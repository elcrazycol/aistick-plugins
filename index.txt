# Репозиторий плагинов AIstickC.
#
# Манифест панели плагинов: что можно поставить на стик.
# Файл плагина лежит рядом с этим манифестом.

name = switch
description = Включить, выключить или переключить выключатель
params = channel=1|2|3|all, state*=on|off|toggle
file = switch.ais
version = 1

name = weather
description = Погода на сегодня (быстрее интернет-поиска)
params = city
file = weather.ais
version = 2

name = lamp
description = Лампа на GPIO26
params = state*=on|off|toggle
file = lamp.ais
version = 1

name = show_text
description = Показать текст на экране стика
params = text*
file = show_text.ais
version = 1

name = web_search
description = Найти свежую информацию в интернете и ответить по ней
params = query*
file = web_search.ais
version = 2
